######## 1: docker cp einer Liste in ein Root-eigenes Verzeichnis (/etc)
    Host-Datei: -rw-r--r--@ <Host-Benutzer geschwärzt>
docker cp rc=0
    -rw-r--r-- 501 dialout /etc/u-boot-egress.allow
    vscode schreibt in /etc/u-boot-egress.allow: nein
    vscode loescht /etc/u-boot-egress.allow: nein
    Datei noch da
######## 2: docker cp einer Liste in ein Verzeichnis des Benutzers (/var/lib/u-boot/status, gehoert vscode)
docker cp rc=0
    -rw-r--r-- 501 dialout /var/lib/u-boot/status/egress.allow
    vscode schreibt die Datei: nein
    vscode loescht und legt neu an: rc=0
    -rw-r--r-- vscode vscode /var/lib/u-boot/status/egress.allow
