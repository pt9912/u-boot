# Devcontainer image for u-boot-messung-amd64-pod — generated by `u-boot generate devcontainer`.
FROM mcr.microsoft.com/devcontainers/base:debian

# Non-root default user (LH-FA-DEV-004).
USER root
# Nested rootless Podman (LH-FA-DEV-007): the static release archive of
# mgoltzsche/podman-static (Podman, crun, conmon, netavark, aardvark-dns,
# passt, fuse-overlayfs). Version, checksums and signing key are pinned here,
# in one place; the build stops before the extraction unless the SHA-256 and a
# signature of the pinned primary key match. `uidmap` keeps the setuid
# newuidmap/newgidmap from Debian.
ARG PODMAN_VERSION=6.1.4
ARG PODMAN_SHA256_AMD64=9dd51073fac5a1742f88cf3caff5554615cb4d67ba554d25e8807be635547af5
ARG PODMAN_SHA256_ARM64=c106d09d7524209ce01eb3a02a7e621e9c55ec31d0fada48439cf77aa07afdb6
ARG PODMAN_SIGNING_KEY=0CCF102C4F95D89E583FF1D4F8B5AF50344BB503
RUN apt-get update \
 && apt-get install -y --no-install-recommends uidmap ca-certificates \
 && rm -rf /var/lib/apt/lists/* \
 && arch="$(dpkg --print-architecture)" \
 && case "$arch" in amd64) sha256="${PODMAN_SHA256_AMD64}" ;; arm64) sha256="${PODMAN_SHA256_ARM64}" ;; *) echo "podman-static has no archive for $arch" >&2; exit 1 ;; esac \
 && work="$(mktemp -d)" \
 && cd "$work" \
 && url="https://github.com/mgoltzsche/podman-static/releases/download/v${PODMAN_VERSION}/podman-linux-${arch}.tar.gz" \
 && curl -fsSLO "$url" \
 && curl -fsSLO "$url.asc" \
 && echo "${sha256}  podman-linux-${arch}.tar.gz" | sha256sum -c - \
 && curl -fsSL "https://keys.openpgp.org/vks/v1/by-fingerprint/${PODMAN_SIGNING_KEY}" -o key.asc \
 && export GNUPGHOME="$work/gnupg" \
 && install -d -m 0700 "$GNUPGHOME" \
 && gpg --batch --import key.asc \
 && status="$(gpg --batch --status-fd 1 --verify "podman-linux-${arch}.tar.gz.asc" "podman-linux-${arch}.tar.gz")" \
 && test "$(echo "$status" | awk '$2 == "VALIDSIG" {print $12}')" = "${PODMAN_SIGNING_KEY}" \
 && tar -xzf "podman-linux-${arch}.tar.gz" -C / --strip-components=1 --no-same-owner \
 && cd / \
 && rm -rf "$work" /README.md \
 && printf "unqualified-search-registries = ['docker.io']\n" > /etc/containers/registries.conf \
 && ln -sf /usr/local/bin/podman /usr/local/bin/docker \
 && (grep -q '^vscode:' /etc/subuid || echo 'vscode:100000:65536' >> /etc/subuid) \
 && (grep -q '^vscode:' /etc/subgid || echo 'vscode:100000:65536' >> /etc/subgid) \
 && mkdir -p /home/vscode/.local/share/containers \
 && chown -R vscode:vscode /home/vscode/.local
COPY sandbox-init.sh /usr/local/bin/u-boot-sandbox-init
# Ollama (LH-FA-DEV-003): the CPU-only release archive of ollama/ollama.
# Version and checksums are pinned here, in one place; the build stops before
# the extraction unless the SHA-256 matches. The upstream publishes no
# signature: the check catches a release asset that was swapped later, not a
# release that was harmful from the start. The GPU libraries stay out of the
# image; `zstd` is installed for the extraction and removed in the same step.
# The image starts no server and sets no listen address. The model store
# belongs to the container user, so the named volume mounted there inherits
# that ownership.
ARG OLLAMA_VERSION=0.40.6
ARG OLLAMA_SHA256_AMD64=c164ab8b1b28e2a247c29719b8e404c423c5b83ae64c3ab4e59d50c7ff7fc5e0
ARG OLLAMA_SHA256_ARM64=0a279d9428ad17b06fdb7dddf3b059b2c148433585fa3b30676a102fc459bb9f
RUN apt-get update \
 && apt-get install -y --no-install-recommends zstd \
 && rm -rf /var/lib/apt/lists/* \
 && arch="$(dpkg --print-architecture)" \
 && case "$arch" in amd64) sha256="${OLLAMA_SHA256_AMD64}" ;; arm64) sha256="${OLLAMA_SHA256_ARM64}" ;; *) echo "ollama has no archive for $arch" >&2; exit 1 ;; esac \
 && work="$(mktemp -d)" \
 && cd "$work" \
 && curl -fsSLO "https://github.com/ollama/ollama/releases/download/v${OLLAMA_VERSION}/ollama-linux-${arch}.tar.zst" \
 && echo "${sha256}  ollama-linux-${arch}.tar.zst" | sha256sum -c - \
 && tar --zstd -xf "ollama-linux-${arch}.tar.zst" -C /usr/local --no-same-owner --exclude='lib/ollama/cuda_*' --exclude='lib/ollama/vulkan' \
 && cd / \
 && rm -rf "$work" \
 && apt-get purge -y --auto-remove zstd \
 && mkdir -p /home/vscode/.ollama \
 && chown vscode:vscode /home/vscode/.ollama

=== .devcontainer/egress-init.sh ===
# BEGIN U-BOOT MANAGED BLOCK: init
# Egress restriction (LH-FA-DEV-008, ADR-0012), run by `postStartCommand`
# via `sudo sh`. A guardrail, not a sandbox boundary. Generated by
# `u-boot generate devcontainer`; extend the allowlist with
# `u-boot config set devcontainer.sandbox.egress.allow <host>`.
#
# Mechanism: a local dnsmasq resolves only the allowed names and adds the
# answers to an nftables set; everything else is dropped by nftables.
# Policy devcontainer.sandbox.onUnavailable: warn
on_unavailable="warn"
hosts="api.github.com auth.docker.io codeload.github.com dd20bb891979d25aebc8bec07b2b3bbc.r2.cloudflarestorage.com deb.debian.org ghcr.io github.com objects.githubusercontent.com production.cloudflare.docker.com production.cloudfront.docker.com raw.githubusercontent.com registry-1.docker.io registry.ollama.ai security.debian.org"
status_dir=/var/lib/u-boot/status

# Writes the status file `u-boot doctor` reads: ok | fallback | failed.
# The file is replaced as a whole; a write error never changes the exit code.
write_status() {
  [ -d "$status_dir" ] || return 0
  tmp="$status_dir/.egress.$$"
  { printf 'u-boot-status 1\nresult=%s\nat=%s\n' "$1" "$(date -u +%Y-%m-%dT%H:%M:%SZ)" > "$tmp" && mv -f "$tmp" "$status_dir/egress"; } 2>/dev/null || rm -f "$tmp"
  return 0
}

# Without the NET_ADMIN capability nftables cannot be used.
if ! nft list tables >/dev/null 2>&1; then
  echo "u-boot sandbox: egress restriction unavailable (NET_ADMIN not granted); consider restricting egress at network or DNS level" >&2
  if [ "$on_unavailable" = "fail" ]; then
    write_status failed
    exit 11
  fi
  write_status fallback
  exit 0
fi

# Keep the original resolver configuration (restart-safe).
[ -f /etc/resolv.conf.u-boot-orig ] || cp /etc/resolv.conf /etc/resolv.conf.u-boot-orig
upstreams=$(awk '/^nameserver/ && $2 !~ /:/ {print $2}' /etc/resolv.conf.u-boot-orig)
pidfile=/run/u-boot-egress-dnsmasq.pid
conf=/etc/u-boot-egress-dnsmasq.conf

# Re-run safe: stop a previous resolver, drop the previous tables.
if [ -f "$pidfile" ]; then kill "$(cat "$pidfile")" 2>/dev/null || true; fi
nft delete table ip u_boot_egress 2>/dev/null || true
nft delete table ip6 u_boot_egress6 2>/dev/null || true

{
  echo "listen-address=127.0.0.1"
  echo "bind-interfaces"
  echo "no-resolv"
  echo "no-hosts"
  echo "pid-file=$pidfile"
  for h in $hosts; do
    for ns in $upstreams; do echo "server=/$h/$ns"; done
    echo "nftset=/$h/4#ip#u_boot_egress#allowed"
  done
} > "$conf"

nft add table ip u_boot_egress
nft add set ip u_boot_egress allowed '{ type ipv4_addr; flags timeout; timeout 1h; }'
nft add chain ip u_boot_egress output '{ type filter hook output priority 0; policy drop; }'
nft add rule ip u_boot_egress output oifname lo accept
nft add rule ip u_boot_egress output ct state established,related accept
nft add rule ip u_boot_egress output ip daddr @allowed accept
for ns in $upstreams; do
  nft add rule ip u_boot_egress output ip daddr "$ns" udp dport 53 accept
  nft add rule ip u_boot_egress output ip daddr "$ns" tcp dport 53 accept
done
nft add table ip6 u_boot_egress6
nft add chain ip6 u_boot_egress6 output '{ type filter hook output priority 0; policy drop; }'
nft add rule ip6 u_boot_egress6 output oifname lo accept

dnsmasq --conf-file="$conf"
echo "nameserver 127.0.0.1" > /etc/resolv.conf
write_status ok
# END U-BOOT MANAGED BLOCK: init
=== .devcontainer/sandbox-init.sh ===
# BEGIN U-BOOT MANAGED BLOCK: init
# Startup check for nested rootless Podman (LH-FA-DEV-007), run by
# `postCreateCommand` via `sh`. Generated by `u-boot generate devcontainer`.
# Policy devcontainer.sandbox.onUnavailable: warn
on_unavailable="warn"
status_dir=/var/lib/u-boot/status

# Writes the status file `u-boot doctor` reads: ok | fallback | failed.
# The file is replaced as a whole; a write error never changes the exit code.
write_status() {
  [ -d "$status_dir" ] || return 0
  tmp="$status_dir/.vfs.$$"
  { printf 'u-boot-status 1\nresult=%s\nat=%s\n' "$1" "$(date -u +%Y-%m-%dT%H:%M:%SZ)" > "$tmp" && mv -f "$tmp" "$status_dir/vfs"; } 2>/dev/null || rm -f "$tmp"
  return 0
}

unavailable() {
  echo "u-boot sandbox: $1" >&2
  if [ "$on_unavailable" = "fail" ]; then
    write_status failed
    exit 11
  fi
  write_status fallback
}

# Nested user namespaces (blocked by Seccomp/AppArmor => Podman cannot run).
if ! unshare --user --map-root-user true 2>/dev/null; then
  echo "u-boot sandbox: nested user namespaces are blocked (Seccomp/AppArmor; on Ubuntu 24.04+ hosts also kernel.apparmor_restrict_unprivileged_userns=1); rootless Podman cannot run" >&2
  exit 11
fi

# /dev/fuse: without it fuse-overlayfs fails, fall back to vfs storage. The
# user drop-in overrides the overlay driver of the system drop-in
# (/etc/containers/storage.conf.d/00-podman-static.conf) and drops its
# mount_program. `podman build` mounts a kernel overlay over the build context
# whose upperdir lies in image_copy_tmp_dir; that directory must not sit on the
# container's overlay root file system, so it lives in the storage volume.
if [ ! -c /dev/fuse ]; then
  unavailable "/dev/fuse is not available; falling back to vfs storage"
  conf_dir="$HOME/.config/containers"
  copy_tmp_dir="$HOME/.local/share/containers/tmp"
  mkdir -p "$conf_dir/storage.conf.d" "$copy_tmp_dir"
  printf '[storage]\ndriver = "vfs"\n\n[storage.options.overlay]\nmount_program = ""\n' > "$conf_dir/storage.conf.d/99-u-boot-vfs.conf"
  printf '[engine]\nimage_copy_tmp_dir = "%s"\n' "$copy_tmp_dir" > "$conf_dir/containers.conf"
else
  write_status ok
fi
# END U-BOOT MANAGED BLOCK: init
