# BEGIN U-BOOT MANAGED BLOCK: init
# Devcontainer image for u-boot-messung-amd64-dokn — generated by `u-boot generate devcontainer`.
FROM mcr.microsoft.com/devcontainers/base:debian

# Non-root default user (LH-FA-DEV-004).
USER root
# Nested Docker daemon (LH-FA-DEV-007): packages from the Docker APT repository
# (the key file must hold exactly the pinned primary key), daemon started at
# container start.
RUN install -m 0755 -d /etc/apt/keyrings \
 && curl -fsSL https://download.docker.com/linux/debian/gpg -o /etc/apt/keyrings/docker.asc \
 && test "$(gpg --show-keys --with-colons /etc/apt/keyrings/docker.asc | awk -F: '/^pub/ {p=1} /^sub/ {p=0} /^fpr/ && p {print $10}')" = "9DC858229FC7DD38854AE2D88D81803C0EBFCD88" \
 && echo "deb [arch=$(dpkg --print-architecture) signed-by=/etc/apt/keyrings/docker.asc] https://download.docker.com/linux/debian $(. /etc/os-release && echo "$VERSION_CODENAME") stable" > /etc/apt/sources.list.d/docker.list \
 && apt-get update \
 && apt-get install -y --no-install-recommends docker-ce docker-ce-cli containerd.io docker-buildx-plugin docker-compose-plugin iptables ca-certificates \
 && rm -rf /var/lib/apt/lists/* \
 && usermod -aG docker vscode
COPY docker-init.sh /usr/local/bin/u-boot-docker-init
# Egress restriction (LH-FA-DEV-008).
RUN apt-get update \
 && apt-get install -y --no-install-recommends dnsmasq-base nftables \
 && rm -rf /var/lib/apt/lists/*
COPY egress-init.sh /usr/local/bin/u-boot-egress-init
# Status files of the container start scripts (LH-FA-DEV-007); `u-boot doctor`
# reads them from the host. The directory belongs to the container user so the
# `postCreateCommand` and the `sudo` scripts can write there.
RUN mkdir -p /var/lib/u-boot/status && chown vscode:vscode /var/lib/u-boot/status
# Sandbox profile (LH-FA-DEV-006): workspace lives in a named volume that
# inherits ownership from this directory.
RUN mkdir -p "/workspaces/u-boot-messung-amd64-dokn" && chown vscode:vscode "/workspaces/u-boot-messung-amd64-dokn"
# Git credentials come from $GIT_TOKEN at runtime (LH-FA-DEV-009); nothing is stored in the image.
RUN git config --system credential.helper '!f() { test -n "$GIT_TOKEN" && printf "username=x-access-token\npassword=%s\n" "$GIT_TOKEN"; }; f'
USER vscode

=== .devcontainer/docker-init.sh ===
# BEGIN U-BOOT MANAGED BLOCK: init
# Starts the nested Docker daemon (LH-FA-DEV-007), run by
# `postStartCommand` via `sudo sh`. Generated by `u-boot generate devcontainer`.
# The daemon storage (/var/lib/docker) is a named volume per instance.
# Policy devcontainer.sandbox.onUnavailable: warn
on_unavailable="warn"
log=/var/log/u-boot-dockerd.log
pidfile=/var/run/docker.pid
wait_seconds=60
status_dir=/var/lib/u-boot/status

# Writes the status file `u-boot doctor` reads: ok | fallback | failed.
# The file is replaced as a whole; a write error never changes the exit code.
write_status() {
  [ -d "$status_dir" ] || return 0
  tmp="$status_dir/.daemon.$$"
  { printf 'u-boot-status 1\nresult=%s\nat=%s\n' "$1" "$(date -u +%Y-%m-%dT%H:%M:%SZ)" > "$tmp" && mv -f "$tmp" "$status_dir/daemon"; } 2>/dev/null || rm -f "$tmp"
  return 0
}

unavailable() {
  echo "u-boot sandbox: $1" >&2
  if [ "$on_unavailable" = "fail" ]; then
    write_status failed
    exit 11
  fi
  write_status fallback
  exit 0
}

ready() { timeout 5 docker info >/dev/null 2>&1; }

if ready; then
  write_status ok
  exit 0
fi

# A daemon that is still starting keeps its PID file; a stopped container
# keeps the PID and socket files of the killed daemon.
pid=$(cat "$pidfile" 2>/dev/null)
if [ -z "$pid" ] || [ "$(cat "/proc/$pid/comm" 2>/dev/null)" != "dockerd" ]; then
  rm -f "$pidfile" /var/run/docker.sock /var/run/docker/containerd/containerd.pid
  # The daemon needs the privileges of `--privileged` (iptables, cgroups, mounts).
  setsid nohup dockerd --host=unix:///var/run/docker.sock --group docker >"$log" 2>&1 &
  pid=$!
fi

deadline=$(($(date +%s) + wait_seconds))
while [ "$(date +%s)" -lt "$deadline" ]; do
  if ready; then
    write_status ok
    exit 0
  fi
  if ! kill -0 "$pid" 2>/dev/null; then
    unavailable "the nested Docker daemon exited at start (container started without --privileged?); no daemon available, see $log"
  fi
  sleep 1
done
unavailable "the nested Docker daemon is not ready after ${wait_seconds}s; no daemon available, see $log"
# END U-BOOT MANAGED BLOCK: init
=== .devcontainer/egress-init.sh ===
# BEGIN U-BOOT MANAGED BLOCK: init
# Egress restriction (LH-FA-DEV-008, ADR-0012), run by `postStartCommand`
# via `sudo sh`. A guardrail, not a sandbox boundary. Generated by
# `u-boot generate devcontainer`; extend the allowlist with
# `u-boot config set devcontainer.sandbox.egress.allow <host>`.
#
# Mechanism: a local dnsmasq resolves only the allowed names and adds the
# answers to an nftables set; everything else is dropped by nftables.
# Policy devcontainer.sandbox.onUnavailable: warn
on_unavailable="warn"
hosts="api.github.com auth.docker.io codeload.github.com deb.debian.org download.docker.com ghcr.io github.com objects.githubusercontent.com production.cloudflare.docker.com production.cloudfront.docker.com raw.githubusercontent.com registry-1.docker.io security.debian.org"
status_dir=/var/lib/u-boot/status

# Writes the status file `u-boot doctor` reads: ok | fallback | failed.
# The file is replaced as a whole; a write error never changes the exit code.
write_status() {
  [ -d "$status_dir" ] || return 0
  tmp="$status_dir/.egress.$$"
  { printf 'u-boot-status 1\nresult=%s\nat=%s\n' "$1" "$(date -u +%Y-%m-%dT%H:%M:%SZ)" > "$tmp" && mv -f "$tmp" "$status_dir/egress"; } 2>/dev/null || rm -f "$tmp"
  return 0
}

# Without the NET_ADMIN capability nftables cannot be used.
if ! nft list tables >/dev/null 2>&1; then
  echo "u-boot sandbox: egress restriction unavailable (NET_ADMIN not granted); consider restricting egress at network or DNS level" >&2
  if [ "$on_unavailable" = "fail" ]; then
    write_status failed
    exit 11
  fi
  write_status fallback
  exit 0
fi

# Keep the original resolver configuration (restart-safe).
[ -f /etc/resolv.conf.u-boot-orig ] || cp /etc/resolv.conf /etc/resolv.conf.u-boot-orig
upstreams=$(awk '/^nameserver/ && $2 !~ /:/ {print $2}' /etc/resolv.conf.u-boot-orig)
pidfile=/run/u-boot-egress-dnsmasq.pid
conf=/etc/u-boot-egress-dnsmasq.conf

# Re-run safe: stop a previous resolver, drop the previous tables.
if [ -f "$pidfile" ]; then kill "$(cat "$pidfile")" 2>/dev/null || true; fi
nft delete table ip u_boot_egress 2>/dev/null || true
nft delete table ip6 u_boot_egress6 2>/dev/null || true

{
  echo "listen-address=127.0.0.1"
  echo "bind-interfaces"
  echo "no-resolv"
  echo "no-hosts"
  echo "pid-file=$pidfile"
  for h in $hosts; do
    for ns in $upstreams; do echo "server=/$h/$ns"; done
    echo "nftset=/$h/4#ip#u_boot_egress#allowed"
  done
} > "$conf"

nft add table ip u_boot_egress
nft add set ip u_boot_egress allowed '{ type ipv4_addr; flags timeout; timeout 1h; }'
nft add chain ip u_boot_egress output '{ type filter hook output priority 0; policy drop; }'
nft add rule ip u_boot_egress output oifname lo accept
nft add rule ip u_boot_egress output ct state established,related accept
nft add rule ip u_boot_egress output ip daddr @allowed accept
for ns in $upstreams; do
  nft add rule ip u_boot_egress output ip daddr "$ns" udp dport 53 accept
  nft add rule ip u_boot_egress output ip daddr "$ns" tcp dport 53 accept
done
nft add table ip6 u_boot_egress6
nft add chain ip6 u_boot_egress6 output '{ type filter hook output priority 0; policy drop; }'
nft add rule ip6 u_boot_egress6 output oifname lo accept

dnsmasq --conf-file="$conf"
echo "nameserver 127.0.0.1" > /etc/resolv.conf
write_status ok
# END U-BOOT MANAGED BLOCK: init
